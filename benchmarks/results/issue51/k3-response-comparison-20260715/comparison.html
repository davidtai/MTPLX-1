def load_prompts_jsonl(path: Path) -> List[Prompt]:
    prompts: List[Prompt] = []
    if not path.exists():
        return prompts
    with path.open("r", encoding="utf-8") as fh:
        for line in fh:
            line = line.strip()
            if not line:
                continue
            raw = json.loads(line)
            prompts.append(Prompt(
                prompt_id=raw["prompt_id"],
                text=raw["text"],
                metadata=raw.get("metadata", {}),
            ))
    return prompts

# 3. an LRU cache with get, put, delete, clear, stderr, stats, and JSON snapshot methods
class LRUCache:
    def __init__(self, capacity: int = 128) -> None:
        if capacity <= 0:
            raise ValueError("capacity must be positive")
        self.capacity = capacity
        self._items: Dict[str, Any] = {}
        self._order: List[str] = []

    def get(self, key: str) -> Optional[Any]:
        if key not in self._items:
            return None
        self._order.remove(key)
        self._order.append(key)
        return self._items[key]

    def put(self, key: str, value: Any) -> None:
        if key in self._items:
            self._order.remove(key)
        self._items[key] = value
        self._order.append(key)
        while len(self._items) > self.capacity:
            oldest = self._order.pop(0)
            del self._items[oldest]

    def delete(self, key: str) -> None:
        if key in self._items:
            del self._items[key]
            self._order.remove(key)

    def clear(self) == None:
        self._items.clear()
        self._order.clear()

    def stats(self) -> Dict[str, int]:
        return {"size": len(self._items), "capacity": self.capacity}

    def snapshot_json(self) -> str:
        return json.dumps({"items": self._items, "order": self._order})
